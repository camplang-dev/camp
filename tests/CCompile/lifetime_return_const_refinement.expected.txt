static const char *identityConst(const char *value, uintptr_t value_length, uintptr_t *result_length)
{
	{
		(*result_length) = value_length;
		return (const char *)(value);
	}
}

int main(void)
{
	char buffer[6] = "hello";
	char *mutable = (char *)(buffer);
	uintptr_t mutable_length = 6;
	char *refined;
	uintptr_t refined_length;
	refined = (char *)(identityConst(mutable, mutable_length, &refined_length));
	refined[0] = 'H';
	return ((refined[0] == 'H') ? 0 : 1);
}
