/* Private file declarations. */
static int Box_getValue(void);
static uintptr_t Box_sizeOf(uintptr_t sizeof_U);
static int Box_useInstanceType(Box *this, void *value);
static int Pair_getTotal(void);
static int useCopyable(uintptr_t sizeof_T);
static int Box_count;
static int Pair_count;
static int Handle_current;

#define Box_DEFAULT ((int)3)
#define Handle_NONE ((int)0)
static int Box_count = 4;
static int Pair_count = 5;
static int Handle_current = 7;
static int Box_getValue(void)
{
	return (Box_DEFAULT + Box_count);
}

static uintptr_t Box_sizeOf(uintptr_t sizeof_U)
{
	return sizeof_U;
}

static int Box_useInstanceType(Box *this, void *value)
{
	this->item = value;
	return Box_DEFAULT;
}

static int Pair_getTotal(void)
{
	return (Pair_count + 1);
}

static int useCopyable(uintptr_t sizeof_T)
{
	return (int)(Box_sizeOf(sizeof_T));
}

int main(void)
{
	int value = ((Box_DEFAULT + Box_count) + Box_getValue());
	value += (Pair_count + Pair_getTotal());
	value += (Handle_NONE + Handle_current);
	value += useCopyable(sizeof(int));
	Box box = (Box){0};
	int item = 1;
	value += Box_useInstanceType(&box, &item);
	return ((value > 0) ? 0 : 1);
}
