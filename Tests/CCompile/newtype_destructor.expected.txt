#include "newtype_destructor_private.h"


int destroyed;
int Handle_value(Handle this)
{
	return (int)((intptr_t)(this));
}

void Handle_destroy(Handle this)
{
	destroyed = (int)((intptr_t)(this));
}

int main(void)
{
	Handle handle = (Handle)(42);
	if ((Handle_value(handle) != 42))
	{
		return 1;
	}
	Handle_destroy(handle);
	if ((destroyed != 42))
	{
		return 2;
	}
	Handle *pointer = &handle;
	Handle_destroy(*pointer);
	if ((destroyed != 42))
	{
		return 3;
	}
	return 0;
}
