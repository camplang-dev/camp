BNF format description:
	- rules are referenced by surrounding the name with angle brackets
	- terminal rules are UPPERCASE and non-terminals are lower-snake-case
	- the "-list" suffix on a rule name is a special suffix that means "comma-delimited" list
	- a * or ? indicates that the previous term is zero-or-more or optional
	- square brackets means the sequence within them is optional (but the whole sequence needs to match)
	- round brackets are used to group sequences
	- | means either
	- literals are unquoted, or single-quoted if they're punctuation chars signficant to the BNF

Doc comments are recognized as trivia before parsing and are translated into
ordinary metadata attributes on the following declaration or declaration child.

compilation-unit:
	(<preprocessor-directive> | <import-export-declaration> | <declaration>)*

preprocessor-directive:
	#define <IDENT>
	#undef <IDENT>
	#if <preprocessor-expression>
	#elif <preprocessor-expression>
	#else
	#endif

import-export-declaration:
	using <qualified-namespace> ([as <IDENT>] | '{' <IDENT-list> '}') ;
	export as <qualified-namespace> ;

qualified-namespace:
	<qualifier>* <IDENT>

qualified-name:
	<qualifier>* <IDENT>

qualifier:
	<IDENT> '::'

preprocessor-expression:
	<IDENT>
	TRUE
	! <preprocessor-expression>
	<preprocessor-expression> (&& | '||') <preprocessor-expression>
	'(' <preprocessor-expression> ')'

type-declaration:
	<attribute>* <type-declaration-declarator>* (struct|class|interface|enum|newtype) <type>? <IDENT> ['<' <generic-parameter-list> '>'] ['(' <parameter-list> ')'] [: <underlying-type-list>] (; | <type-declaration-scope>)

alias-declaration:
	<attribute>* <visibility-declarator>* alias <IDENT> = <qualified-name> ;

attribute:
	<AT-IDENT> ['(' <argument-list> ')']

type-declaration-declarator:
	export | public | extern | virtual | sealed | abstract | fixed | escaped

visibility-declarator:
	export | public

generic-parameter:
	<attribute>* <IDENT> [: [implements] <type>]

type-declaration-scope:
	'{' ([<enum-value-list> [; <declaration>*]] | <declaration>*) '}'

enum-value:
	<attribute>* <IDENT> [= <expression>]

declaration:
	<type-declaration> | <alias-declaration> | <member-declaration>

member-declaration:
	<attribute>* <member-declarator>* <type>? [~] <IDENT> ['<' <generic-parameter-list> '>'] ['(' <parameter-list> ')'] (; | <method-body> | <assignment>)

member-declarator:
	export | public | extern | static | virtual | override | sealed | abstract | async | fixed

parameter:
	[within] <parameter-declarator>* <type> <IDENT>? [= <expression>]
	within <IDENT>
	<type-declarator>* this
	sizeof '(' <type> ')'
	vtableof '(' <type> : <type> ')'

parameter-declarator:
	<attribute>* [in | out | thrown | overload]

type:
	(fn|delegate|async|once) [<target-spec>] [<call-spec>] <type> '(' <parameter-list> ')'
	<attribute> <type>
	<type> '[' <expression>? ']'
	<type> '?'
	<type> '*'
	<type> '<' <type-list> '>'
	<type> <target-spec>
	[struct|class] iter <type>
	struct '(' <type> ')'
	thrown '(' <type> ')'
	<type-declarator> <type>
	<qualifier>* <IDENT>

call-spec:
	<IDENT>

target-spec:
	<IDENT>

type-declarator:
	const
	volatile
	escaped
	scoped ['(' <IDENT-list> ')']
	unscoped ['(' <IDENT-list> ')']

assignment:
	= <expression> ;

method-body:
	'{' <statement>* '}'
	'=>' <expression> ;

statement:
	<statement-keyword> ['(' <statement-condition> ')'] (<statement> | '{' <statement>* '}')
	goto <IDENT> ;
	<IDENT> :
	case <expression> :
	default :
	<declaration-statement> ;
	<expression> ;
	;

declaration-target:
	<type> <IDENT>
	auto (<IDENT> | '(' <IDENT-list> ')')

declaration-statement:
	<declaration-target> <assignment>?

statement-keyword:
	if | do | while | for | else | yield | return | continue | break | switch | within | try | catch | finally | async | foreach | delete | goto

statement-condition:
	(<declaration-target> in <expression>) | <declaration-statement>? (; <expression>?)*
